Explanation

Purpose. YLLO2D2 is yllo2d.f for the second OPAL92 table, at a different $Z$ ("DBG 5/94 modified to use second opacity table at different Z"): the same method and equations, with the COMMON blocks GLLOT2, LLOT2, LLOT42 and LINTPL2.

How it fits. Called by YLLO3D2 and LL4TH; calls FINDEX and INTPOL. Its coefficients come from YLLOC (CFORD2) and LL4TH (CFORD42).

Questions for the experts

  1. Duplicated code. YLLO2D and YLLO2D2 differ only in their COMMON blocks; passing the table as arguments would make one routine. Worth doing in the overhaul? (The stop message differs too, "OPAL95 2D2 CHECK NDSS", with the same mislabel as YLLO2D's: these are OPAL92 tables.)

7C     TWO DIMENSIONAL INTERPOLATION FOR OPACITY
8C     AND THE DERIVTIVES.
9C DBG 5/94 Modified to use second opacity table at different Z
10C M1, M2, AND M3 ARE NEAREST GRID POINT OF ABUNDANCE, TEMPERATURE,
11C AND DENSITY WE GOT AT 'GETD'.
12C O IS OPACITY.
13C OL IS DLOG(O)
14C QODF IS THE PARTIAL DERIVATIVE OF O WRT D.
15C QOTF IS THE PARTIAL DERIVATIVE OF O WRT T.
16C =========================================================
17      IMPLICIT REAL*8 (A-H,O-Z)
18      IMPLICIT LOGICAL*4 (L)
19      PARAMETER (NUMT=50)
20      PARAMETER (NUMD=17)
21      PARAMETER (NUMX=3)
22      PARAMETER (NUMXT=NUMT*NUMX)
23      PARAMETER (NUM4D=4*NUMD)
24      COMMON /GLLOT2/GRIDT2(NUMT),GRIDX2(NUMX),GRHOT32(NUMD)
25      COMMON /LLOT2/OPACITY2(NUMXT,NUMD),NUMXM2,NUMTM2
26      COMMON /LLOT42/XENV42,ZENV42,CFORD42(NUMT,NUM4D),KIPM12
27      COMMON /LINTPL2/ CFORD2(NUMXT,NUM4D),NDS2(NUMXT),NDD2(NUMXT)
28      REAL*8 XT(NUMT),YTO(NUMT)
29      REAL*8 AQOD(NUMT)
30      INTEGER JT,IT,ITS,ITF
31      LOGICAL*4 LMORE
32      SAVE
33      LMORE=.TRUE.
34CCC FOR SIX GRID POINTS OF TEMPERATURE
35      ITS=M2-2
36      IF(ITS.LE.0)ITS=1
37      ITF=M2+3
38      IF(ITF.GT.NUMTM2)ITF=NUMTM2
39      IF(M1.LT.4)THEN
40          MM1=M1
41       ELSE
42          MM1=KIPM12
43      ENDIF
44      JT=0
45      DO 300 IT=ITS,ITF
46        INDEX1=IT+(MM1-1)*NUMTM2
47        NDSS=NDS2(INDEX1)
48        IF(NDSS.NE.1)STOP ' OPAL95 2D2 CHECK NDSS '
49        NDF=NDSS+NDD2(INDEX1)-1
50        CALL FINDEX(GRHOT32,NDF,D,M3)
51        IF(M3.LT.0)THEN
52C  OUT SIDE THEN  LINEAR EXTRAPOLATION
53          M3=-M3
54          KNOT=M3-NDSS+1
55          INDEX2=4*(KNOT-1)
56          DX=D-GRHOT32(M3)
57          IF(M1.LT.4)THEN
58            C1=CFORD2(INDEX1,INDEX2+1)
59            C2=CFORD2(INDEX1,INDEX2+2)
60          ELSE
61            C1=CFORD42(IT,INDEX2+1)
62            C2=CFORD42(IT,INDEX2+2)
63          ENDIF
64          OL0=C2*DX+C1
65          QODI=C2
66        ELSE
67C  IN SIDE THEN  SPLINE INTERPOLATION
68          KNOT=M3-NDSS+1
69          INDEX2=4*(KNOT-1)
70          DX=D-GRHOT32(M3)
71          IF(M1.LT.4)THEN
72            C1=CFORD2(INDEX1,INDEX2+1)
73            C2=CFORD2(INDEX1,INDEX2+2)
74            C3=CFORD2(INDEX1,INDEX2+3)
75            C4=CFORD2(INDEX1,INDEX2+4)
76          ELSE
77            C1=CFORD42(IT,INDEX2+1)
78            C2=CFORD42(IT,INDEX2+2)
79            C3=CFORD42(IT,INDEX2+3)
80            C4=CFORD42(IT,INDEX2+4)
81          ENDIF
82C INTERPOLATION FOR OPACITY(OL) IN THE ENTRY D AND THE EACH T-GRID
83C ESTIMATES THE PARTIAL DERIVATIVE OF OL WRT D
84C EVALUATES THE INTERPOLATION VALUE IN THE SUB-RANGE WE DETERMINED.
85          OL0=((C4*DX+C3)*DX+C2)*DX+C1
86          QODI=(3.0D0*C4*DX+2.0D0*C3)*DX+C2
87        ENDIF
88C
89        JT=JT+1
90        XT(JT)=GRIDT2(IT)
91        YTO(JT)=OL0
92        AQOD(JT)=QODI
93  300 CONTINUE
94      IF(XT(1).GT.T.OR.XT(JT).LT.T) STOP ' EXTRAPOLATION FAILS '
95CC INTERPOLATION FOR THE OPACITY IN THE ENTRY T AND D.
96CC GET THE PARTIAL DERIVATIVE OF OL WRT T.
97      CALL INTPOL(XT,YTO,JT,T,OL00,QOTF)
98      OL=OL00
99      O=10.0D0**OL
100C QOTF = D LN(O)/D LN(T)
101CC FIND THE PARTIAL DERIVATIVE VALUE OF OL WRT D IN THE GIVEN T AND D
102      CALL INTPOL(XT,AQOD,JT,T,QODF,QODTI)
103C QODF = D LN(O)/D LN(D)
104      RETURN
105      END
